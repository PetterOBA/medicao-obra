// ============================================================ // MEDIÇÃO DE OBRA — MACRO-ETAPAS (Supremo Goiânia / Supremo Goiás) // Google Apps Script — tudo via GET pra evitar bloqueio de CORS // ============================================================ const ABA = "Medicoes"; function garantirAba() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(ABA); if (!sheet) { sheet = ss.insertSheet(ABA); sheet.appendRow(["Empreendimento","Unidade","CodigoEtapa","NomeEtapa","Status","AtualizadoEm"]); sheet.setFrozenRows(1); sheet.getRange(1,1,1,6).setBackground("#0a1338").setFontColor("#ffffff").setFontWeight("bold"); sheet.setColumnWidths(1,6,140); } return sheet; } function resposta(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // Acha o índice da linha (1-based, já contando header) pra uma combinação // Empreendimento + Unidade + CodigoEtapa. Retorna -1 se não existir. function acharLinha(data, emp, unidade, codigo) { for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(emp) && String(data[i][1]) === String(unidade) && String(data[i][2]) === String(codigo)) { return i; // índice no array "data" (0-based, linha real = i+1) } } return -1; } function gravarUm(sheet, data, emp, unidade, codigo, nome, status) { const agora = new Date().toISOString(); const idx = acharLinha(data, emp, unidade, codigo); if (idx === -1) { sheet.appendRow([emp, unidade, codigo, nome, status, agora]); data.push([emp, unidade, codigo, nome, status, agora]); // mantém "data" atualizado pra próximas buscas no mesmo request } else { const linhaReal = idx + 1; sheet.getRange(linhaReal, 5).setValue(status); sheet.getRange(linhaReal, 6).setValue(agora); data[idx][4] = status; data[idx][5] = agora; } } function doGet(e) { try { const sheet = garantirAba(); const action = e.parameter.action; // ── SALVAR UM ITEM ── if (action === "save") { const data = sheet.getDataRange().getValues(); gravarUm(sheet, data, e.parameter.emp, e.parameter.unidade, e.parameter.codigo, e.parameter.nome, e.parameter.status); return resposta({ success: true }); } // ── SALVAR VÁRIOS ITENS DE UMA VEZ (cascata de sub-etapas) ── if (action === "saveBatch") { const data = sheet.getDataRange().getValues(); const items = JSON.parse(e.parameter.items || "[]"); items.forEach(function(it) { gravarUm(sheet, data, e.parameter.emp, e.parameter.unidade, it.codigo, it.nome, it.status); }); return resposta({ success: true, count: items.length }); } // ── CARREGAR (opcionalmente filtrado por empreendimento) ── if (action === "load") { const data = sheet.getDataRange().getValues(); if (data.length <= 1) return resposta({ success: true, data: [] }); const emp = e.parameter.emp; const rows = []; for (let i = 1; i < data.length; i++) { if (emp && String(data[i][0]) !== String(emp)) continue; rows.push({ Empreendimento: data[i][0], Unidade: String(data[i][1]), CodigoEtapa: data[i][2], NomeEtapa: data[i][3], Status: data[i][4] }); } return resposta({ success: true, data: rows }); } return resposta({ success: false, error: "Ação desconhecida" }); } catch (err) { return resposta({ success: false, error: String(err) }); } }
